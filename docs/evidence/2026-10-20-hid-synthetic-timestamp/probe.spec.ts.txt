/**
 * بارکدخوان USB/HID (keyboard wedge) روی صندوق — رویدادهای واقعی کیبورد، نه `fill`.
 *
 * شکایت مالک: «اسکن بارکد چاپی فقط 2 در فیلد می‌گذارد». بارکد داخلی EAN-13 با پیشوند
 * ۲۰ است. `ScanBuffer` عمداً رقم **اول** را عبور می‌دهد (هنوز از تایپ انسان قابل تشخیص
 * نیست) و از رقم دوم به بعد را می‌بلعد؛ کامنتش می‌گفت «فراخوان همان یکی را پاک کند»، ولی
 * صندوق هرگز پاکش نمی‌کرد. پس «2» در کادر جست‌وجو می‌ماند. پسوند Tab پذیرفته نمی‌شد؛
 * اسکنر باید پسوند Enter یا Tab داشته باشد (ثبت خودکار بی‌پسوند عمداً نیست).
 *
 * هر کلید با `keyboard.down`/`up` واقعی و بی‌تأخیر فرستاده می‌شود (مثل اسکنر)؛ تایپ دستی
 * با فاصلهٔ انسانی. مقدار بارکد هرگز تغییر نمی‌کند: همان رشته به `POST …/scan` می‌رسد.
 */

/**
 * مثل اسکنر: همهٔ کلیدها پشت سر هم و بی مکث. در Chromium رویداد واقعی سیستم‌عامل با CDP
 * (keyDown با متن، پس رقم اول واقعاً در کادر می‌نشیند) و همه با هم فرستاده می‌شوند: `await`
 * جدا برای هر کلید زیر بار CI از آستانهٔ ۵۰ms کندتر می‌شد و آزمون دیگر اسکنر نبود.
 * موتورهای دیگر CDP ندارند: رویداد keydown در یک تکهٔ کد صفحه (نویسه‌ای درج نمی‌شود، پس
 * آن‌جا فقط تشخیص، پسوند و مقدار بارکد سنجیده می‌شود؛ نشت رقم اول را Chromium می‌سنجد).
 * ⚠️ `timeStamp` رویداد ساخته‌شده زمانِ **ساختن** آن است، نه رسیدن کلید. همهٔ رویدادها اول
 *    ساخته و بعد فرستاده می‌شوند؛ ساختن هر کدام پس از فرستادن قبلی یعنی هر توقف میزبان
 *    (CI با بار ۲۰ روی ۴ هسته) فاصلهٔ کلید ۱ و ۲ را از ۵۰ms رد می‌کرد و بافر رقم اول را
 *    دور می‌انداخت: «2345678901234567» در PR180 (webkit-dark-768). در Chromium ورودی واقعی
 *    (CDP) زمانش را هنگام رسیدن گرفت و کندی صفحه آن را بزرگ نکرد؛ WebKit بومی سنجیده نشده
 *    است (docs/evidence/2026-10-18-hid-timestamp-probe).
 */
async function scanCdp(page: Page, keys: string[], browserName: string) {
  if (browserName === "chromium") {
    const cdp = await page.context().newCDPSession(page);
    const code = (k: string) => {
      const ascii = /^[0-9]$/.test(k) ? k : /^[\u06F0-\u06F9]$/.test(k) ? String(k.charCodeAt(0) - 0x06f0) : null;
      if (ascii !== null) return { key: k, text: k, code: `Digit${ascii}`, windowsVirtualKeyCode: 48 + Number(ascii) };
      return k === "Enter" ? { key: "Enter", text: "\r", code: "Enter", windowsVirtualKeyCode: 13 } : { key: "Tab", code: "Tab", windowsVirtualKeyCode: 9 };
    };
    await Promise.all(keys.flatMap(k => { const c = code(k); const { text: _t, ...up } = c as { text?: string };
      return [cdp.send("Input.dispatchKeyEvent", { type: "keyDown", ...c }), cdp.send("Input.dispatchKeyEvent", { type: "keyUp", ...up })]; }));
    await cdp.detach();
  } else {
    await page.evaluate(ks => { const t = document.activeElement ?? document.body;
      const events = ks.map(key => new KeyboardEvent("keydown", { key, bubbles: true, cancelable: true }));
      for (const e of events) t.dispatchEvent(e); }, keys);
  }
}

type Mode = "old-stall" | "new-stall" | "new-slow";
async function synthetic(page: Page, keys: string[], mode: Mode) {
  return page.evaluate(([ks, m]) => {
    const t = document.activeElement ?? document.body;
    const stall = () => { const end = performance.now() + 60; while (performance.now() < end) { /* توقف میزبان */ } };
    const t0 = performance.now();
    const events = ks.map((key, i) => {
      if (m !== "new-slow" && i === 2) stall();           // توقف میان ساختن کلید ۲ و ۳
      const e = new KeyboardEvent("keydown", { key, bubbles: true, cancelable: true });
      if (m !== "old-stall") Object.defineProperty(e, "timeStamp", { value: t0 + i * 2 + (m === "new-slow" && i >= 2 ? 60 : 0) });
      return e;
    });
    for (const e of events) t.dispatchEvent(e);
    return events.slice(0, 4).map(e => Math.round(e.timeStamp * 10) / 10);
  }, [keys, mode] as const);
}
const LEGACY = "12345678901234567";
for (const mode of ["old-stall", "new-stall", "new-slow"] as const) {
  test(`probe ${mode}`, async ({ page, api }) => {
    const { scans } = await open(page, api);
    await page.locator("body").click({ position: { x: 1, y: 1 } });
    const ts = await synthetic(page, [...LEGACY, "Enter"], mode);
    await expect.poll(() => scans.length, { timeout: 3000 }).toBeGreaterThanOrEqual(0);
    await page.waitForTimeout(300);
    console.log(`PROBE ${mode}: timeStamps[0..3]=${JSON.stringify(ts)} gap2→3=${(ts[2]! - ts[1]!).toFixed(1)}ms scans=${JSON.stringify(scans)}`);
  });
}
